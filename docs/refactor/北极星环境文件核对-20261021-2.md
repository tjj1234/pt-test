```bash
sudo awk -F= 'NF < 2 {print "格式错误：第 " NR " 行"; bad=1; next} /[<>]/ {print "仍有占位符：第 " NR " 行（" $1 "）"; bad=1} END {if (!NR) print "环境文件为空"; else if (!bad) print "格式/占位符检查通过，行数：" NR}' /etc/northstar/northstar.env
```
